**Goal:** Replace `all-you-plan-api` with a clean-room GraphQL API for the new "all you plan" domain — server, JWT auth, nine domains plus a derived stats query, and seed data.

- [ ] **Step 2: `src/config/auth.ts`** — JWT sign/verify and refresh-cookie helpers.

```ts
import jwt from 'jsonwebtoken'
import type { Response } from 'express'
import { env, isProd } from './env.js'

export interface TokenPayload { userId: string; tokenVersion: number }
const REFRESH_COOKIE = 'ayp_refresh'

export function signAccessToken(p: TokenPayload): string {
  return jwt.sign(p, env.JWT_SECRET, { expiresIn: env.JWT_EXPIRES_IN })
}
export function signRefreshToken(p: TokenPayload): string {
  return jwt.sign(p, env.JWT_REFRESH_SECRET, { expiresIn: env.JWT_REFRESH_EXPIRES_IN })
}
export function verifyAccessToken(t: string): TokenPayload {
  return jwt.verify(t, env.JWT_SECRET) as TokenPayload
}
export function verifyRefreshToken(t: string): TokenPayload {
  return jwt.verify(t, env.JWT_REFRESH_SECRET) as TokenPayload
}
export function setRefreshCookie(res: Response, token: string): void {
  res.cookie(REFRESH_COOKIE, token, {
    httpOnly: true, path: '/graphql', sameSite: 'lax', secure: isProd,
    maxAge: 7 * 24 * 60 * 60 * 1000,
  })
}
export function clearRefreshCookie(res: Response): void {
  res.clearCookie(REFRESH_COOKIE, { path: '/graphql' })
}
export const REFRESH_COOKIE_NAME = REFRESH_COOKIE
```

- [ ] **Step 3: `src/middleware/auth.ts`** — context builder and `requireAuth`.

```ts
import { GraphQLError } from 'graphql'
import type { Request, Response } from 'express'
import { verifyAccessToken } from '../config/auth.js'
import { User } from '../domains/auth/model.js'

export interface Context { userId: string | null; req?: Request; res?: Response }

export async function buildContext(req?: Request, res?: Response): Promise<Context> {
  const header = req?.headers.authorization
  if (!header?.startsWith('Bearer ')) return { userId: null, req, res }
  try {
    const payload = verifyAccessToken(header.slice(7))
    const user = await User.findById(payload.userId).select('tokenVersion')
    if (!user || user.tokenVersion !== payload.tokenVersion) return { userId: null, req, res }
    return { userId: payload.userId, req, res }
  } catch {
    return { userId: null, req, res }
  }
}

export function requireAuth(ctx: Context): string {
  if (!ctx.userId) throw new GraphQLError('Not authenticated', { extensions: { code: 'UNAUTHENTICATED' } })
  return ctx.userId
}
```

import { z } from 'zod'
import { GraphQLError } from 'graphql'

export function validate<T>(schema: z.ZodType<T>, input: unknown): T {
  const result = schema.safeParse(input)
  if (!result.success) {
    throw new GraphQLError(result.error.issues[0]?.message ?? 'Invalid input', {
      extensions: { code: 'BAD_USER_INPUT' },
    })
  }
  return result.data
}

export const registerInput = z.object({
  email: z.string().email(),
  password: z.string().min(8).regex(/[a-z]/).regex(/[A-Z]/).regex(/[0-9]/),
  name: z.string().min(1).max(80),
})

- [ ] **Step 5: `src/domains/auth/service.ts`** — register/login/refresh/logout/reset logic.

Implement these exported async functions, each returning `{ user: UserDoc; accessToken: string; refreshToken: string }` where noted:
- `registerUser(input)` — validate with `registerInput`; reject duplicate email with `GraphQLError` code `CONFLICT`; create the `User` (the model hashes the password); issue both tokens.
- `loginUser(email, password)` — find the user `.select('+password')`; `comparePassword`; on failure throw `GraphQLError` code `UNAUTHENTICATED` "Invalid credentials"; issue both tokens.
- `refreshSession(refreshToken)` — `verifyRefreshToken`; load user; check `tokenVersion`; re-issue both tokens.
- `logoutUser(userId)` — `$inc tokenVersion` by 1.
- `forgotPassword(email)` — set a `crypto.randomUUID()` `resetToken` + 1h `resetTokenExpiry`; log it (no email service); always resolve (anti-enumeration).
- `resetPassword(token, newPassword)` — find by valid unexpired `resetToken`; set password; `$inc tokenVersion`; clear reset fields.

Token issuance: `signAccessToken/signRefreshToken({ userId: user.id, tokenVersion: user.tokenVersion })`.

```ts
import { readFileSync } from 'node:fs'
import { fileURLToPath } from 'node:url'
import { createServer } from 'node:http'
import express from 'express'
import cors from 'cors'
import helmet from 'helmet'
import cookieParser from 'cookie-parser'
import { ApolloServer } from '@apollo/server'
import { expressMiddleware } from '@apollo/server/express4'
import { makeExecutableSchema } from '@graphql-tools/schema'
import { mergeTypeDefs, mergeResolvers } from '@graphql-tools/merge'
import { WebSocketServer } from 'ws'
import { useServer } from 'graphql-ws/lib/use/ws'
import { env } from './config/env.js'
import { logger } from './config/logger.js'
import { connectDatabase } from './config/database.js'
import { buildContext } from './middleware/auth.js'
import { authResolvers } from './domains/auth/resolvers.js'

function sdl(domain: string): string {
  return readFileSync(fileURLToPath(new URL(`./domains/${domain}/schema.graphql`, import.meta.url)), 'utf8')
}

// As domains are added in later tasks, register their SDL + resolvers here.
const DOMAINS = ['auth']
const typeDefs = mergeTypeDefs(DOMAINS.map(sdl))
const resolvers = mergeResolvers([authResolvers])

async function start(): Promise<void> {
  await connectDatabase()
  const schema = makeExecutableSchema({ typeDefs, resolvers })
  const app = express()
  const httpServer = createServer(app)
  const wsServer = new WebSocketServer({ server: httpServer, path: '/graphql' })
  const wsCleanup = useServer({ schema, context: async () => ({ userId: null }) }, wsServer)

  const apollo = new ApolloServer({
    schema,
    plugins: [
      {
        async serverWillStart() {
          return { async drainServer() { await wsCleanup.dispose() } }
        },
      },
    ],
  })
  await apollo.start()

  app.use(helmet({ crossOriginEmbedderPolicy: false }))
  app.use(cors({ origin: env.CORS_ORIGIN, credentials: true }))
  app.use(cookieParser())
  app.get('/health', (_req, res) => { res.json({ status: 'ok' }) })
  app.use('/graphql', express.json(), expressMiddleware(apollo, {
    context: async ({ req, res }) => buildContext(req, res),
  }))

  httpServer.listen(Number(env.PORT), () => {
    logger.info(`API ready at http://localhost:${env.PORT}/graphql`)
  })
}

start()
```

Note: `DOMAINS`, `mergeTypeDefs`, and `mergeResolvers` are extended in every later domain task — each task adds its domain name to `DOMAINS` and its resolver map to the `mergeResolvers([...])` array, and imports its resolver module.

- [ ] **Step 2: `service.ts`** — `loadToday(userId, date)`:
  1. Call `materializeChoreOccurrences(userId, date)` (from the chores service) so the day's due chores exist as Tasks.
  2. Load all `Task`s with `scheduledDate` on `date`, sorted by `scheduledTime`.
  3. Compute `kpis`: `streak` from `User.streak.current`; `todayDone`/`todayTotal` from the task list; `focusMinutes` = sum of `effortMinutes` over done tasks; `activeProjects` = count of non-archived Projects with `status != idle`.
  4. `sunrise`/`sunset` are fixed strings `'05:12'` / `'21:46'` (static — see spec Non-goals).
  5. Return `{ date, sunrise, sunset, kpis, tasks }`.

- [ ] **Step 3: `resolvers.ts`** — `todayResolvers`: `Query.today` — `requireAuth`, default `date` to today (server local date `YYYY-MM-DD`), delegate to `loadToday`.

- [ ] **Step 5: Test** `src/__tests__/today.test.ts` — unit-test the KPI computation: given a fixed array of task objects, the `computeKpis` helper returns the expected `todayDone`/`todayTotal`/`focusMinutes`. Extract `computeKpis(tasks, user, projectCount)` as a pure function and test it.
