      sunset
      kpis { streak todayDone todayTotal focusMinutes activeProjects }

        id title note scheduledTime done completedAt effortMinutes tag projectId order
      }
    }
  }
`

  mutation CompleteTask($id: ID!) {
    completeTask(id: $id) { id done completedAt }
  }
`

  }
`

### Task 2: Today store

- [ ] **Step 1: Create `src/stores/today.store.ts`**

```ts
import { create } from 'zustand'
import { apolloClient } from '@/api/apollo'
import { TODAY_QUERY, COMPLETE_TASK, MOVE_UNFINISHED } from '@/api/operations/today'

export interface TodayTask {
  id: string
  title: string
  note?: string | null
  scheduledTime?: string | null
  done: boolean
  effortMinutes?: number | null
  tag?: string | null
  projectId?: string | null
}
export interface TodayKpis {
  streak: number; todayDone: number; todayTotal: number; focusMinutes: number; activeProjects: number
}
export interface TodayView {
  date: string; sunrise: string; sunset: string; kpis: TodayKpis; tasks: TodayTask[]
}

interface TodayState {
  view: TodayView | null
  loading: boolean
  error: string
  load: (date?: string) => Promise<void>
  completeTask: (id: string) => Promise<void>
  moveUnfinished: () => Promise<void>
}

export const useTodayStore = create<TodayState>((set, get) => ({
  view: null,
  loading: false,
  error: '',
  load: async (date) => {
    set({ loading: true, error: '' })
    try {
      const { data } = await apolloClient.query({
        query: TODAY_QUERY,
        variables: { date: date ?? null },
        fetchPolicy: 'network-only',
      })
      set({ view: data.today, loading: false })
    } catch (e) {
      set({ error: (e as Error).message, loading: false })
    }
  },
  completeTask: async (id) => {
    await apolloClient.mutate({ mutation: COMPLETE_TASK, variables: { id } })
    await get().load(get().view?.date)
  },
  moveUnfinished: async () => {
    const view = get().view
    if (!view) return
    await apolloClient.mutate({ mutation: MOVE_UNFINISHED, variables: { fromDate: view.date } })
    await get().load(view.date)
  },
}))
```

### Task 4: Today compound components

**Files:** Create `src/components/today/{KpiTile,KpiRow,TaskRow,CoachCard,NextCard}.tsx`.

- [ ] **Step 1: `KpiTile.tsx`** — a tile: `label` (small muted), `value` (serif 26), optional `unit` (small). Uses `useTheme()` + `StyleSheet`, matching the primitive style conventions.

```tsx
import React from 'react'
import { View, Text } from 'react-native'
import { useTheme } from '@/theme/useTheme'

interface KpiTileProps { label: string; value: string; unit?: string }

export function KpiTile({ label, value, unit }: KpiTileProps) {
  const { palette, fonts } = useTheme()
  return (
    <View style={{ flex: 1, gap: 2, borderRadius: 16, padding: 14, backgroundColor: palette.paper2 }}>
      <Text style={{ color: palette.muted, fontSize: 11.5, fontFamily: fonts.sansMedium }}>{label}</Text>
      <Text style={{ color: palette.ink, fontSize: 26, fontFamily: fonts.serif }}>{value}</Text>
      {unit ? <Text style={{ color: palette.muted, fontSize: 11, fontFamily: fonts.sans }}>{unit}</Text> : null}
    </View>
  )
}
```

- [ ] **Step 2: `KpiRow.tsx`** — a 3-tile row (Streak / Today / Focus) from a `kpis` prop:

```tsx
import React from 'react'
import { View } from 'react-native'
import { KpiTile } from './KpiTile'
import type { TodayKpis } from '@/stores/today.store'

export function KpiRow({ kpis }: { kpis: TodayKpis }) {
  const focus = `${Math.floor(kpis.focusMinutes / 60)}h ${kpis.focusMinutes % 60}m`
  return (
    <View style={{ flexDirection: 'row', gap: 10 }}>
      <KpiTile label="Streak" value={String(kpis.streak)} unit="days" />
      <KpiTile label="Today" value={`${kpis.todayDone}/${kpis.todayTotal}`} unit="done" />
      <KpiTile label="Focus" value={focus} unit="logged" />
    </View>
  )
}
```
